> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add manual-mode one-click SSH initialization for remote repositories without persisting passwords, private key contents, or passphrases.

**Architecture:** Add a focused main-process SSH initialization module and expose only three controlled `remote.*` RPC procedures. The renderer adds a small initialization panel inside `AddRemoteRepositoryDialog`; all durable remote repository state remains unchanged.

**Tech Stack:** Electron main process, React renderer, TypeScript, Valibot RPC schemas, Vitest, OpenSSH command-line tools, `execa`, and `node-pty`.

**Repository Constraint:** This repository's AGENTS instructions say not to plan or execute git commits unless the user asks. This plan intentionally omits commit steps.

```ts
import crypto from 'node:crypto'
import { promises as fs } from 'node:fs'
import os from 'node:os'
import path from 'node:path'
import { execa } from 'execa'
import * as pty from 'node-pty'
import type {
  SshHostKeyConfirmation,
  SshInitAccessInput,
  SshInitConnectionInput,
  SshInitKeyStatus,
  SshInitPrepareResult,
  SshInitTrustHostKeyInput,
} from '#/shared/remote-repo.ts'
import type { ExecResult } from '#/shared/git-types.ts'

const SSH_INIT_TIMEOUT_MS = 30_000
const SSH_INIT_PASSWORD_TIMEOUT_MS = 45_000
const SSH_DIR_MODE = 0o700
const SSH_PRIVATE_KEY_MODE = 0o600
const SSH_PUBLIC_KEY_MODE = 0o644

interface HostKeyParts {
  hosts: string
  keyType: string
  key: string
  line: string
}

interface KeyPaths {
  sshDir: string
  privateKey: string
  publicKey: string
  knownHosts: string
}

interface PasswordCommandInvocation {
  command: string
  args: string[]
}

type CommandRunner = typeof execa
type PasswordCommandRunner = (
  invocation: PasswordCommandInvocation,
  password: string,
  options?: { signal?: AbortSignal; timeoutMs?: number },
) => Promise<ExecResult>

interface InitOptions {
  signal?: AbortSignal
  homeDir?: string
  run?: CommandRunner
  runPasswordCommand?: PasswordCommandRunner
}

export async function prepareSshInit(
  input: SshInitConnectionInput,
  options: InitOptions = {},
): Promise<SshInitPrepareResult> {
  try {
    const run = options.run ?? execa
    const paths = keyPaths(options.homeDir)
    const keyStatus = await ensureDefaultKey(paths, run, options.signal)
    const hostStatus = await inspectHostKey(input.host, input.port, paths, run, options.signal)
    if (hostStatus.status === 'trusted') return { ok: true, keyStatus, hostKeyStatus: 'trusted' }
    if (hostStatus.status === 'needs-confirmation') {
      return { ok: true, keyStatus, hostKeyStatus: 'needs-confirmation', confirmation: hostStatus.confirmation }
    }
    return { ok: false, keyStatus, hostKeyStatus: 'changed', message: 'error.ssh-host-key-changed' }
  } catch (err) {
    return { ok: false, message: safeErrorMessage(err) }
  }
}

export async function trustSshHostKey(
  input: SshInitTrustHostKeyInput,
  options: InitOptions = {},
): Promise<ExecResult> {
  try {
    const parsed = parseHostKeyLine(input.key)
    if (!parsed) return { ok: false, message: 'error.invalid-arguments' }
    const fingerprint = fingerprintHostKey(parsed)
    if (fingerprint !== input.fingerprint) return { ok: false, message: 'error.ssh-host-key-mismatch' }
    const paths = keyPaths(options.homeDir)
    await ensureSshDir(paths)
    const line = `${knownHostName(input.host, input.port)} ${parsed.keyType} ${parsed.key}\n`
    await fs.appendFile(paths.knownHosts, line, 'utf-8')
    return { ok: true, message: 'ssh host key trusted' }
  } catch (err) {
    return { ok: false, message: safeErrorMessage(err) }
  }
}

export async function initializeSshAccess(
  input: SshInitAccessInput,
  options: InitOptions = {},
): Promise<ExecResult> {
  const run = options.run ?? execa
  const runPasswordCommand = options.runPasswordCommand ?? runPasswordPtyCommand
  const paths = keyPaths(options.homeDir)
  try {
    await ensureDefaultKey(paths, run, options.signal)
    const publicKey = (await fs.readFile(paths.publicKey, 'utf-8')).trim()
    if (!publicKey) return { ok: false, message: 'error.ssh-public-key-missing' }
    const invocation = await buildInstallInvocation(input, paths.publicKey, publicKey, run, options.signal)
    const result = await runPasswordCommand(invocation, input.password, {
      signal: options.signal,
      timeoutMs: SSH_INIT_PASSWORD_TIMEOUT_MS,
    })
    return result.ok ? result : { ...result, message: redactSshInitOutput(result.message || result.stderr || '', input.password) }
  } catch (err) {
    return { ok: false, message: redactSshInitOutput(safeErrorMessage(err), input.password) }
  }
}

export function knownHostName(host: string, port: number): string {
  return port === 22 ? host : `[${host}]:${port}`
}

export function parseHostKeyLine(line: string): HostKeyParts | null {
  const trimmed = line.trim()
  if (!trimmed || trimmed.startsWith('#')) return null
  const fields = trimmed.split(/\s+/)
  const offset = fields[0]?.startsWith('@') ? 1 : 0
  const hosts = fields[offset]
  const keyType = fields[offset + 1]
  const key = fields[offset + 2]
  if (!hosts || !keyType || !key || !isSshKeyType(keyType)) return null
  return { hosts, keyType, key, line: `${hosts} ${keyType} ${key}` }
}

export function fingerprintHostKey(parts: HostKeyParts): string {
  const digest = crypto.createHash('sha256').update(Buffer.from(parts.key, 'base64')).digest('base64').replace(/=+$/, '')
  return `SHA256:${digest}`
}

export function redactSshInitOutput(text: string, password: string): string {
  const withoutPassword = password ? text.split(password).join('[redacted]') : text
  return withoutPassword.replace(/(password:\s*).*/gi, '$1[redacted]').trim()
}

function keyPaths(homeDir = os.homedir()): KeyPaths {
  const sshDir = path.join(homeDir, '.ssh')
  return {
    sshDir,
    privateKey: path.join(sshDir, 'id_ed25519'),
    publicKey: path.join(sshDir, 'id_ed25519.pub'),
    knownHosts: path.join(sshDir, 'known_hosts'),
  }
}

async function ensureSshDir(paths: KeyPaths): Promise<void> {
  await fs.mkdir(paths.sshDir, { recursive: true, mode: SSH_DIR_MODE })
  await fs.chmod(paths.sshDir, SSH_DIR_MODE)
}

async function pathExists(filePath: string): Promise<boolean> {
  try {
    await fs.access(filePath)
    return true
  } catch {
    return false
  }
}

async function ensureDefaultKey(paths: KeyPaths, run: CommandRunner, signal?: AbortSignal): Promise<SshInitKeyStatus> {
  await ensureSshDir(paths)
  const privateExists = await pathExists(paths.privateKey)
  const publicExists = await pathExists(paths.publicKey)
  if (privateExists && publicExists) return 'existing'
  if (privateExists) {
    const { stdout } = await run('ssh-keygen', ['-y', '-f', paths.privateKey], {
      cancelSignal: signal,
      timeout: SSH_INIT_TIMEOUT_MS,
    })
    await fs.writeFile(paths.publicKey, `${stdout.trim()}\n`, { mode: SSH_PUBLIC_KEY_MODE })
    return 'public-key-recreated'
  }
  await run('ssh-keygen', ['-t', 'ed25519', '-f', paths.privateKey, '-N', '', '-C', `goblin@${os.hostname()}`], {
    cancelSignal: signal,
    timeout: SSH_INIT_TIMEOUT_MS,
  })
  await fs.chmod(paths.privateKey, SSH_PRIVATE_KEY_MODE)
  return 'generated'
}

async function inspectHostKey(
  host: string,
  port: number,
  paths: KeyPaths,
  run: CommandRunner,
  signal?: AbortSignal,
): Promise<
  | { status: 'trusted' }
  | { status: 'needs-confirmation'; confirmation: SshHostKeyConfirmation }
  | { status: 'changed' }
> {
  const scanned = await scanHostKeys(host, port, run, signal)
  const known = await readKnownHostKeys(paths, host, port)
  if (known.length === 0) {
    const first = scanned[0]
    if (!first) return { status: 'changed' }
    return {
      status: 'needs-confirmation',
      confirmation: { host, port, key: first.line, keyType: first.keyType, fingerprint: fingerprintHostKey(first) },
    }
  }
  const scannedFingerprints = new Set(scanned.map(fingerprintHostKey))
  return known.some((key) => scannedFingerprints.has(fingerprintHostKey(key))) ? { status: 'trusted' } : { status: 'changed' }
}

async function scanHostKeys(host: string, port: number, run: CommandRunner, signal?: AbortSignal): Promise<HostKeyParts[]> {
  const { stdout } = await run('ssh-keyscan', ['-p', String(port), '-T', '10', host], {
    cancelSignal: signal,
    timeout: SSH_INIT_TIMEOUT_MS,
  })
  return stdout
    .split(/\r?\n/)
    .map(parseHostKeyLine)
    .filter((value): value is HostKeyParts => value !== null)
}

async function readKnownHostKeys(paths: KeyPaths, host: string, port: number): Promise<HostKeyParts[]> {
  try {
    const content = await fs.readFile(paths.knownHosts, 'utf-8')
    const name = knownHostName(host, port)
    return content
      .split(/\r?\n/)
      .map(parseHostKeyLine)
      .filter((value): value is HostKeyParts => value !== null && value.hosts.split(',').includes(name))
  } catch {
    return []
  }
}

async function buildInstallInvocation(
  input: SshInitAccessInput,
  publicKeyPath: string,
  publicKey: string,
  run: CommandRunner,
  signal?: AbortSignal,
): Promise<PasswordCommandInvocation> {
  const destination = `${input.user}@${input.host}`
  if (await commandAvailable('ssh-copy-id', run, signal)) {
    return {
      command: 'ssh-copy-id',
      args: ['-o', 'StrictHostKeyChecking=yes', '-p', String(input.port), '-i', publicKeyPath, destination],
    }
  }
  const script = [
    'umask 077',
    'mkdir -p "$HOME/.ssh"',
    'touch "$HOME/.ssh/authorized_keys"',
    `grep -qxF ${shellQuote(publicKey)} "$HOME/.ssh/authorized_keys" || printf '%s\\n' ${shellQuote(
      publicKey,
    )} >> "$HOME/.ssh/authorized_keys"`,
    'chmod 700 "$HOME/.ssh"',
    'chmod 600 "$HOME/.ssh/authorized_keys"',
  ].join('; ')
  return {
    command: 'ssh',
    args: ['-o', 'StrictHostKeyChecking=yes', '-p', String(input.port), destination, `sh -lc ${shellQuote(script)}`],
  }
}

async function commandAvailable(command: string, run: CommandRunner, signal?: AbortSignal): Promise<boolean> {
  try {
    await run('command', ['-v', command], { shell: true, cancelSignal: signal, timeout: SSH_INIT_TIMEOUT_MS })
    return true
  } catch {
    return false
  }
}

async function runPasswordPtyCommand(
  invocation: PasswordCommandInvocation,
  password: string,
  options: { signal?: AbortSignal; timeoutMs?: number } = {},
): Promise<ExecResult> {
  return new Promise((resolve) => {
    let output = ''
    let settled = false
    const child = pty.spawn(invocation.command, invocation.args, {
      cols: 80,
      rows: 24,
      cwd: os.homedir(),
      env: { ...process.env, TERM: 'xterm-256color' },
    })
    const finish = (result: ExecResult) => {
      if (settled) return
      settled = true
      clearTimeout(timer)
      child.kill()
      resolve(result)
    }
    const timer = setTimeout(() => finish({ ok: false, message: 'timeout' }), options.timeoutMs ?? SSH_INIT_PASSWORD_TIMEOUT_MS)
    options.signal?.addEventListener('abort', () => finish({ ok: false, message: 'cancelled' }), { once: true })
    child.onData((data) => {
      output += data
      if (/password:\s*$/i.test(output)) child.write(`${password}\r`)
    })
    child.onExit(({ exitCode }) => {
      const message = redactSshInitOutput(output, password)
      finish(exitCode === 0 ? { ok: true, message } : { ok: false, message })
    })
  })
}

function isSshKeyType(value: string): boolean {
  return value === 'ssh-ed25519' || value === 'ssh-rsa' || value.startsWith('ecdsa-sha2-')
}

function shellQuote(value: string): string {
  return `'${value.replace(/'/g, `'\\''`)}'`
}

function safeErrorMessage(err: unknown): string {
  if (err instanceof Error) return err.message
  return String(err)
}
```

```tsx
import type { SshHostKeyConfirmation } from '#/shared/remote-repo.ts'

type InitStage = 'idle' | 'preparing' | 'confirm-host-key' | 'password' | 'installing' | 'success' | 'failed'
type AddRemoteMode = 'config' | 'manual'

interface Props {
  mode: AddRemoteMode
  host: string
  user: string
  port: number
  disabled: boolean
  onBusyChange: (busy: boolean) => void
  onInitialized: () => Promise<void> | void
}

export function SshInitializationPanel({ mode, host, user, port, disabled, onBusyChange, onInitialized }: Props) {
  const t = useT()
  const [stage, setStage] = useState<InitStage>('idle')
  const [password, setPassword] = useState('')
  const [confirmation, setConfirmation] = useState<SshHostKeyConfirmation | null>(null)
  const [error, setError] = useState<string | null>(null)
  const commandBusy = stage === 'preparing' || stage === 'installing'
  const initializing = stage !== 'idle' && stage !== 'success' && stage !== 'failed'
  const canStart = mode === 'manual' && host.trim().length > 0 && user.trim().length > 0 && !disabled && !initializing
  const canInstall = password.length > 0 && !disabled && stage === 'password'

  useEffect(() => {
    onBusyChange(initializing)
  }, [initializing, onBusyChange])

  useEffect(() => {
    setStage('idle')
    setPassword('')
    setConfirmation(null)
    setError(null)
  }, [mode, host, user, port])

  if (mode !== 'manual') return null

  async function handleStart() {
    if (!canStart) return
    setStage('preparing')
    setError(null)
    setPassword('')
    try {
      const result = await rpc.remote.prepareSshInit.query({ host: host.trim(), user: user.trim(), port })
      if (!result.ok) {
        setStage('failed')
        setError(result.message)
        return
      }
      if (result.hostKeyStatus === 'needs-confirmation') {
        setConfirmation(result.confirmation)
        setStage('confirm-host-key')
        return
      }
      setStage('password')
    } catch (err) {
      setStage('failed')
      setError(err instanceof Error ? err.message : String(err))
    }
  }

  async function handleTrustHostKey() {
    if (!confirmation) return
    setStage('preparing')
    setError(null)
    try {
      const result = await rpc.remote.trustSshHostKey.mutate({
        host: confirmation.host,
        port: confirmation.port,
        key: confirmation.key,
        fingerprint: confirmation.fingerprint,
      })
      if (!result.ok) {
        setStage('failed')
        setError(result.message)
        return
      }
      setStage('password')
    } catch (err) {
      setStage('failed')
      setError(err instanceof Error ? err.message : String(err))
    }
  }

  async function handleInstall() {
    if (!canInstall) return
    setStage('installing')
    setError(null)
    try {
      const result = await rpc.remote.initializeSshAccess.mutate({
        host: host.trim(),
        user: user.trim(),
        port,
        password,
      })
      setPassword('')
      if (!result.ok) {
        setStage('failed')
        setError(result.message)
        return
      }
      await onInitialized()
      setStage('success')
    } catch (err) {
      setPassword('')
      setStage('failed')
      setError(err instanceof Error ? err.message : String(err))
    }
  }

  return (

            {t('remote.ssh-init-title')}
          </div>
          <div className="mt-1 text-muted-foreground">{t('remote.ssh-init-description')}</div>
        </div>
        <Button type="button" variant="outline" size="sm" disabled={!canStart} onClick={() => void handleStart()}>
          {commandBusy && <Loader2 className="animate-spin" />}
          {t('remote.ssh-init-action')}
        </Button>
      </div>

      {stage === 'preparing' && <div className="mt-2 text-muted-foreground">{t('remote.ssh-init-preparing')}</div>}

      {stage === 'confirm-host-key' && confirmation && (
        <div className="mt-3 rounded-md border border-border bg-background p-3">
          <div className="text-sm font-medium text-foreground">{t('remote.ssh-init-host-key-title')}</div>
          <div className="mt-1 text-muted-foreground">{t('remote.ssh-init-host-key-message')}</div>
          <dl className="mt-2 grid grid-cols-[4rem_1fr] gap-x-2 gap-y-1 font-mono">
            <dt className="text-muted-foreground">host</dt>
            <dd>{confirmation.host}:{confirmation.port}</dd>
            <dt className="text-muted-foreground">type</dt>
            <dd>{confirmation.keyType}</dd>
            <dt className="text-muted-foreground">SHA256</dt>
            <dd className="break-all">{confirmation.fingerprint}</dd>
          </dl>
          <div className="mt-3 flex justify-end">
            <Button type="button" size="sm" onClick={() => void handleTrustHostKey()}>
              {t('remote.ssh-init-trust-host')}
            </Button>
          </div>
        </div>
      )}

          <label className="block text-sm font-medium text-foreground" htmlFor="ssh-init-password">
            {t('remote.ssh-init-password-label')}
          </label>
          <div className="mt-1 flex gap-2">
```
